const url = "https://raw.githubusercontent.com/pcm-dpc/COVID-19/master/dati-json/dpc-covid19-ita-andamento-nazionale.json"; const xhr = new XMLHttpRequest(); xhr.open("GET", url, true); xhr.onload = function() { if (xhr.status >= 200 && xhr.status < 300) { const dati = JSON.parse(xhr.responseText); const colonne = ["data", "stato", "nuovi_positivi", "totale_positivi", "terapia_intensiva", "dimessi_guariti", "deceduti", "totale_casi", "tamponi"]; // Corretto: prendiamo le chiavi dal primo elemento dell'array const chiavi = Object.keys(dati[0]); let intestazione = ""; chiavi.forEach(chiave => { if (colonne.includes(chiave)) intestazione += "" + chiave + ""; }); intestazione += ""; document.getElementById("intestazione").innerHTML = intestazione; let contenuto = ""; dati.slice(-30).reverse().forEach(giorno => { contenuto += ""; chiavi.forEach(chiave => { if (colonne.includes(chiave)) { let valore = giorno[chiave] ?? "-"; if (chiave === "data" && valore !== "-") valore = valore.substring(0, 10); contenuto += "" + valore + ""; } }); contenuto += ""; }); document.getElementById("contenuto").innerHTML = contenuto; document.getElementById("messaggio").textContent = "Ultime 30 rilevazioni disponibili"; } else { document.getElementById("messaggio").textContent = "Errore nel caricamento dei dati"; } }; xhr.onerror = function() { document.getElementById("messaggio").textContent = "Errore nel caricamento dei dati"; }; xhr.send();
